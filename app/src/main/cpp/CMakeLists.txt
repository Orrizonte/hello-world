cmake_minimum_required(VERSION 3.22.1)
project(beast_finger_camera LANGUAGES CXX)
set(SDK_SO "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libglasses.so")
set(SDK_INC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
if(NOT EXISTS "${SDK_SO}")
  message(FATAL_ERROR "Missing Viture SDK libglasses.so")
endif()
add_library(glasses SHARED IMPORTED)
set_target_properties(glasses PROPERTIES IMPORTED_LOCATION "${SDK_SO}")
add_library(beast_cam_bridge SHARED beast_cam_bridge.cpp)
target_include_directories(beast_cam_bridge PRIVATE "${SDK_INC}")
find_library(log-lib log)
target_link_libraries(beast_cam_bridge glasses ${log-lib})
